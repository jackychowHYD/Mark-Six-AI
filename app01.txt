def load_history_df():
    if os.path.exists(EXCEL_PATH):
        try:
            return pd.read_excel(EXCEL_PATH)
        except Exception:
            pass
    if os.path.exists(CSV_PATH):
        try:
            return pd.read_csv(CSV_PATH)
        except Exception:
            pass
    return pd.DataFrame(columns=['DrawNo', 'N1', 'N2', 'N3', 'N4', 'N5', 'N6', 'Special'])

def save_history_df(df):
    df = df.drop_duplicates(subset=['DrawNo'], keep='last').tail(200)
    df.to_csv(CSV_PATH, index=False)
    try:
        df.to_excel(EXCEL_PATH, index=False)
    except Exception:
        pass

ZODIAC_NUMBERS = {z: [] for z in ZODIAC_ORDER}
for num in range(1, 50):
    z_name = ZODIAC_ORDER[(num - 1) % 12]
    ZODIAC_NUMBERS[z_name].append(num)

WUXING_TAILS = {
    "金": [4, 9], "木": [3, 8], "水": [1, 6], "火": [2, 7], "土": [0, 5]
}

# ==========================================
# 3. 預測核心模型
# ==========================================
def get_zodiac_forecast(main_zodiac, wuxing_pref, period):
    san_he = SAN_HE.get(main_zodiac, [])
    liu_he = LIU_HE.get(main_zodiac, "")
    chong = LIU_CHONG.get(main_zodiac, "")
    
    fav_zodiacs = [main_zodiac] + san_he + ([liu_he] if liu_he else [])
    chong_nums = set(ZODIAC_NUMBERS.get(chong, []))
    
    valid_pool = [n for n in range(1, 50) if n not in chong_nums]
    
    df = load_history_df()
    freq = {}
    if len(df) > 0 and period > 0:
        subset = df.tail(min(period, len(df)))
        all_nums = subset[['N1', 'N2', 'N3', 'N4', 'N5', 'N6']].values.flatten()
        freq = pd.Series(all_nums).value_counts().to_dict()
        
    weights = {}
    for num in valid_pool:
        w = 1.0
        for z in fav_zodiacs:
            if num in ZODIAC_NUMBERS.get(z, []):
                w += 1.5
        if wuxing_pref in WUXING_TAILS and (num % 10 in WUXING_TAILS[wuxing_pref]):
            w += 2.0
        if period > 0:
            w += (freq.get(num, 0) * 0.3)
        weights[num] = w
        
    attempts = 0
    while attempts < 2000:
        attempts += 1
        picked = random.choices(list(weights.keys()), weights=list(weights.values()), k=10)
        picked = list(set(picked))
        if len(picked) >= 6:
            candidate = sorted(picked[:6])
            odds = sum(1 for n in candidate if n % 2 != 0)
            bigs = sum(1 for n in candidate if n >= 25)
            if (2 <= odds <= 4) and (2 <= bigs <= 4):
                return candidate, san_he, liu_he, chong
                
    return sorted(random.sample(valid_pool, 6)), san_he, liu_he, chong
